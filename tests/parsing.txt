[test help]
// --help must work without a target.
expect: success
ft: ./ft_nmap --help

[test missing-target]
// Exactly one target source is required.
expect: failure
ft: ./ft_nmap --ports 30 --scan SYN

[test ip-and-file]
// --ip and --file are mutually exclusive.
expect: failure
ft: ./ft_nmap --ip 172.28.0.10 --file lab/targets-v4.txt --ports 30 --scan SYN

[test unknown-flag]
// Unknown options must be rejected.
expect: failure
ft: ./ft_nmap --ip 172.28.0.10 --wat

[test missing-ip-value]
// Missing --ip argument.
expect: failure
ft: ./ft_nmap --ip

[test duplicate-ip]
// Duplicate --ip.
expect: failure
ft: ./ft_nmap --ip 172.28.0.10 --ip 172.28.0.11

[test missing-file-value]
// Missing --file argument.
expect: failure
ft: ./ft_nmap --file

[test empty-target-file]
// A target file containing no valid targets must fail.
expect: failure
ft: ./ft_nmap --file /dev/null --ports 30 --scan SYN

[test missing-ports-value]
// Missing --ports argument.
expect: failure
ft: ./ft_nmap --ip 172.28.0.10 --ports

[test port-zero]
// Port 0 is invalid.
expect: failure
ft: ./ft_nmap --ip 172.28.0.10 --ports 0 --scan SYN

[test port-overflow]
// Port 65536 is invalid.
expect: failure
ft: ./ft_nmap --ip 172.28.0.10 --ports 65536 --scan SYN

[test reversed-port-range]
// Reversed ranges are invalid.
expect: failure
ft: ./ft_nmap --ip 172.28.0.10 --ports 40-30 --scan SYN

[test empty-port-item]
// Empty comma-separated port items are invalid.
expect: failure
ft: ./ft_nmap --ip 172.28.0.10 --ports 30,,31 --scan SYN

[test too-many-ports]
// The project limit is 1024 selected ports.
expect: failure
ft: ./ft_nmap --ip 172.28.0.10 --ports 1-1025 --scan SYN

[test missing-scan-value]
// Missing --scan argument.
expect: failure
ft: ./ft_nmap --ip 172.28.0.10 --ports 30 --scan

[test invalid-scan]
// Unknown scan names are invalid.
expect: failure
ft: ./ft_nmap --ip 172.28.0.10 --ports 30 --scan CONNECT

[test lowercase-scan]
// Scan names are intentionally parsed as the documented uppercase tokens.
expect: failure
ft: ./ft_nmap --ip 172.28.0.10 --ports 30 --scan syn

[test empty-scan-item]
// Malformed scan lists are invalid.
expect: failure
ft: ./ft_nmap --ip 172.28.0.10 --ports 30 --scan SYN,,UDP

[test speedup-negative]
// Negative speedup is invalid.
expect: failure
ft: ./ft_nmap --ip 172.28.0.10 --speedup -1

[test speedup-too-high]
// Maximum speedup is 250.
expect: failure
ft: ./ft_nmap --ip 172.28.0.10 --speedup 251

[test timeout-zero]
// Timeout must be positive.
expect: failure
ft: ./ft_nmap --ip 172.28.0.10 --timeout 0

[test timeout-text]
// Timeout must be numeric.
expect: failure
ft: ./ft_nmap --ip 172.28.0.10 --timeout abc

[test retries-too-high]
// Maximum retries is 10.
expect: failure
ft: ./ft_nmap --ip 172.28.0.10 --retries 11

[test ttl-too-high]
// TTL/Hop Limit maximum is 255.
expect: failure
ft: ./ft_nmap --ip 172.28.0.10 --ttl 256

[test valid-boundaries]
// Valid maximum CLI boundaries are accepted.
expect: success
ft: sudo ./ft_nmap --ip 172.28.0.10 --ports 65535 --scan SYN --speedup 250 --timeout 200 --retries 10 --ttl 255 --no-dns
