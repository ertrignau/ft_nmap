[test concurrent-processes]
// Two simultaneous scanner processes must not cross-match replies.
ft: rm -f /tmp/ft_nmap_corr_a.out /tmp/ft_nmap_corr_b.out; sudo ./ft_nmap --ip 172.28.0.10 --ports 30-79 --scan SYN,ACK --speedup 24 --timeout 500 --retries 0 --no-dns --reason > /tmp/ft_nmap_corr_a.out 2>&1 & p1=$!; sudo ./ft_nmap --ip 172.28.0.10 --ports 30-79 --scan SYN,ACK --speedup 24 --timeout 500 --retries 0 --no-dns --reason > /tmp/ft_nmap_corr_b.out 2>&1 & p2=$!; wait $p1; s1=$?; wait $p2; s2=$?; printf "\n===== process A =====\n"; cat /tmp/ft_nmap_corr_a.out; printf "\n===== process B =====\n"; cat /tmp/ft_nmap_corr_b.out; rm -f /tmp/ft_nmap_corr_a.out /tmp/ft_nmap_corr_b.out; [ $s1 -eq 0 ] && [ $s2 -eq 0 ]
