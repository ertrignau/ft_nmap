// Deterministic validation of the Docker lab baseline.
// Target: 172.28.0.10
//
// TCP:
//   30 = OPEN
//   31 = CLOSED
//   32 = DROP
//   36 = REJECT
//
// UDP:
//   31 = OPEN
//   30 = CLOSED
//   33 = DROP
//   37 = REJECT


[test lab-syn-baseline]
// SYN: OPEN / CLOSED / DROP / REJECT
ft: sudo ./ft_nmap --ip 172.28.0.10 --ports 30,31,32,36 --scan SYN --reason --no-dns
nmap: sudo nmap -Pn -n -sS -p 30,31,32,36 --reason 172.28.0.10


[test lab-ack-baseline]
// ACK: UNFILTERED / UNFILTERED / DROP / REJECT
ft: sudo ./ft_nmap --ip 172.28.0.10 --ports 30,31,32,36 --scan ACK --reason --no-dns
nmap: sudo nmap -Pn -n -sA -p 30,31,32,36 --reason 172.28.0.10


[test lab-null-baseline]
// NULL:
//   OPEN   -> open|filtered
//   CLOSED -> closed
//   DROP   -> open|filtered
//   REJECT -> filtered
ft: sudo ./ft_nmap --ip 172.28.0.10 --ports 30,31,32,36 --scan NULL --reason --no-dns
nmap: sudo nmap -Pn -n -sN -p 30,31,32,36 --reason 172.28.0.10


[test lab-fin-baseline]
// FIN:
//   OPEN   -> open|filtered
//   CLOSED -> closed
//   DROP   -> open|filtered
//   REJECT -> filtered
ft: sudo ./ft_nmap --ip 172.28.0.10 --ports 30,31,32,36 --scan FIN --reason --no-dns
nmap: sudo nmap -Pn -n -sF -p 30,31,32,36 --reason 172.28.0.10


[test lab-xmas-baseline]
// XMAS:
//   OPEN   -> open|filtered
//   CLOSED -> closed
//   DROP   -> open|filtered
//   REJECT -> filtered
ft: sudo ./ft_nmap --ip 172.28.0.10 --ports 30,31,32,36 --scan XMAS --reason --no-dns
nmap: sudo nmap -Pn -n -sX -p 30,31,32,36 --reason 172.28.0.10


[test lab-udp-baseline]
// UDP:
//   31 = OPEN
//   30 = CLOSED
//   33 = DROP
//   37 = REJECT
ft: sudo ./ft_nmap --ip 172.28.0.10 --ports 30,31,33,37 --scan UDP --reason --no-dns
nmap: sudo nmap -Pn -n -sU -p 30,31,33,37 --reason 172.28.0.10
