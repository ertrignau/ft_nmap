[test baseline]
// ACK scan: RST means UNFILTERED, silence/reject means FILTERED.
ft: sudo ./ft_nmap --ip 172.28.0.10 --ports 30-39 --scan ACK --speedup 16 --timeout 400 --retries 0 --no-dns --reason
nmap: sudo nmap -Pn -n -T4 --max-retries 0 --host-timeout 10s -sA -p 30-39 --reason 172.28.0.10
